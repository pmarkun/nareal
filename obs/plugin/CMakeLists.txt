cmake_minimum_required(VERSION 3.22)
project(nareal-obs VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(libobs REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
find_package(obs-frontend-api QUIET)
add_library(nareal-obs MODULE src/plugin.cpp)
target_link_libraries(nareal-obs PRIVATE OBS::libobs Qt6::Widgets Qt6::Network)
if(TARGET OBS::obs-frontend-api)
  target_link_libraries(nareal-obs PRIVATE OBS::obs-frontend-api)
else()
  find_library(OBS_FRONTEND_LIBRARY NAMES obs-frontend-api REQUIRED)
  target_link_libraries(nareal-obs PRIVATE ${OBS_FRONTEND_LIBRARY})
endif()
set_target_properties(nareal-obs PROPERTIES PREFIX "")
if(MSVC)
  target_compile_options(nareal-obs PRIVATE /W4 /utf-8)
else()
  target_compile_options(nareal-obs PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(GNUInstallDirs)
if(WIN32)
  install(TARGETS nareal-obs RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
elseif(APPLE)
  set_target_properties(nareal-obs PROPERTIES BUNDLE TRUE)
  install(TARGETS nareal-obs BUNDLE DESTINATION . LIBRARY DESTINATION .)
else()
  install(TARGETS nareal-obs LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/obs-plugins)
endif()
